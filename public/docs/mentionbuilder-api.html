📝 Complete Example with All Options

fetch('https://wanda-central.vercel.app/api/mentions/generate', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer mb_live_8k2Jx9pQ4vN7sR1tY6wH3mL5fC0aB'
  },
  body: JSON.stringify({
    // Required
    domain: 'marketplacestudio.nl',
    keywords: [
      'marketplace platform',
      'multi-vendor marketplace',
      'e-commerce solution'
    ],
    
    // Optional settings
    mention_count: 10,
    context: 'professional',
    include_backlink: true,
    target_sites: [
      'reddit.com',
      'quora.com',
      'medium.com'
    ],
    
    // Scheduling
    schedule: {
      frequency: 'daily',
      time: '10:00',
      duration_days: 30
    }
  })
})
.then(r => r.json())
.then(result => {
  console.log('Mentions generated:', result.mentions.length);
  console.log('Remaining quota:', result.remaining_quota);
});
